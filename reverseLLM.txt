          logger.info("LLM-driven Analysis Pipeline initialized")

      def run_analysis(self, data_source: str, 
                      save_results: bool = True) -> Dict[str, Any]:
          """
          运行完整的分析流程
          按照LLM倒推的分析步骤执行
          """
          logger.info("Starting LLM-driven analysis pipeline")

          start_time = datetime.now()

          try:
              # Step 1: 数据摄入
              logger.info("Phase 1: Data Ingestion")
              raw_data = self.data_ingestion.load_and_validate_data(data_source)

              # Step 2: 特征提取
              logger.info("Phase 2: Feature Extraction")
              features = self.feature_extraction.extract_features(raw_data)

              # Step 3: 建立基线（首次运行或定期更新）
              if not self.pattern_analysis.baselines:
                  logger.info("Phase 3a: Establishing Baselines")
                  self.pattern_analysis.establish_baselines(features)

              # Step 4: 模式分析
              logger.info("Phase 3b: Pattern Analysis")
              pattern_results = self.pattern_analysis.detect_patterns(features)

              # Step 5: 结果验证
              logger.info("Phase 4: Validation")
              validation_results = self.pattern_analysis.validate_findings(pattern_results)

              # Step 6: 告警评估
              logger.info("Phase 5: Alert Evaluation")
              alerts = self.alerting.evaluate_alerts(pattern_results)

              # 整合结果
              analysis_results = {
                  'metadata': {
                      'analysis_id': f"analysis_{datetime.now().strftime('%Y%m%d_%H%M%S')}",
                      'start_time': start_time.isoformat(),
                      'end_time': datetime.now().isoformat(),
                      'data_source': data_source,
                      'samples_processed': len(raw_data),
                      'features_extracted': len(features.columns),
                      'llm_confidence': self.config.analysis_confidence
                  },
                  'data_summary': {
                      'total_samples': len(raw_data),
                      'time_range': f"{raw_data.index.min()} to {raw_data.index.max()}",
                      'data_quality_score': self._calculate_data_quality(raw_data)
                  },
                  'pattern_analysis': pattern_results,
                  'validation': validation_results,
                  'alerts': alerts,

                      'reasoning_logic': "{stage2_summary}",

          quality_factors.append(completeness)

          # 一致性（简化版）
          consistency = 1 - (df.duplicated().sum() / len(df))
          quality_factors.append(consistency)

          return np.mean(quality_factors)
