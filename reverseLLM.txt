  class ReverseEngineeringAnalyzer:

      async def full_reverse_analysis(self, 
                                      raw_data_sample: str,
                                      data_context: dict) -> dict:
          """
          执行完整的三阶段反向工程分析
          
          Args:
              raw_data_sample: 原始数据样本
              data_context: 数据上下文信息
              
          Returns:
              完整的分析结果和倒推的流程

              raw_data_sample, data_context
          )

          # 阶段2：推理解释
          print("🧠 阶段2: 分析思路解释...")
          stage2_result = await self.stage2_reasoning_explanation(
              stage1_result
          )

          # 阶段3：倒推需求

          stage3_result = await self.stage3_reverse_engineering(
              stage1_result, stage2_result
          )

          final_analysis = self.integrate_analysis_results(
              stage1_result, stage2_result, stage3_result
          )

          print("✅ 反向工程分析完成!")
          return final_analysis

      async def stage1_initial_analysis(self, raw_data: str, context: dict):
          """阶段1：初步分析"""

          prompt = self.prompts['stage1'].format(
              raw_data_sample=raw_data,
              data_type=context.get('data_type', 'security_logs'),
              time_range=context.get('time_range', '24 hours'),
              sample_count=context.get('sample_count', 'unknown'),
              business_context=context.get('business_context', 'security monitoring')
          )

          response = await self.llm.complete(prompt)
          return self.parse_json_response(response)

      async def stage2_reasoning_explanation(self, stage1_results):
          """阶段2：推理解释"""

          prompt = self.prompts['stage2'].format(
              stage1_results=json.dumps(stage1_results, indent=2)
          )

          response = await self.llm.complete(prompt)
          return self.parse_json_response(response)

      async def stage3_reverse_engineering(self, stage1_results, stage2_reasoning):
          """阶段3：倒推工程"""

          prompt = self.prompts['stage3'].format(
              stage1_results=json.dumps(stage1_results, indent=2),
              stage2_reasoning=json.dumps(stage2_reasoning, indent=2)
          )

          response = await self.llm.complete(prompt)
          return self.parse_json_response(response)

      def integrate_analysis_results(self, stage1, stage2, stage3):
          """整合三个阶段的分析结果"""

          return {
              'meta_info': {
                  'analysis_timestamp': datetime.now().isoformat(),
                  'analysis_method': 'reverse_engineering',
                  'stages_completed': 3
              },

              'initial_assessment': stage1,

                  'analysis_pipeline': self.extract_pipeline(stage3),
                  'validation_framework': self.extract_validation(stage3)
              },

                  'areas_of_uncertainty': self.extract_uncertainties(stage2),
                  'validation_priorities': self.prioritize_validations(stage3)
              }
          }

      def generate_implementation_guide(self, analysis_results):
          """基于分析结果生成实施指南"""

          guide = {
              'executive_summary': self.create_executive_summary(analysis_results),
              'technical_requirements': self.extract_tech_requirements(analysis_results),
              'implementation_phases': self.create_implementation_phases(analysis_results),
              'risk_mitigation': self.create_risk_mitigation_plan(analysis_results),
              'success_metrics': self.define_success_metrics(analysis_results)
          }

          return guide

      # 辅助方法
      def parse_json_response(self, response: str) -> dict:
          """解析LLM的JSON响应"""
          try:
              # 清理和解析JSON
              cleaned = response.strip()
              if '```json' in cleaned:
                  json_start = cleaned.find('```json') + 7
                  json_end = cleaned.find('```', json_start)
                  cleaned = cleaned[json_start:json_end].strip()

              return json.loads(cleaned)
          except Exception as e:
              logging.error(f"JSON解析失败: {e}")
              return {'error': 'parsing_failed', 'raw_response': response}
