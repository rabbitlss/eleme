      def load_and_validate_data(self, file_path: str) -> Tuple[pd.DataFrame, pd.Series]:
          """

          Args:

              # 数据验证
              validated_df = self.validator.schema.validate(df)
              logger.info("Data validation passed")

              if 'target' in validated_df.columns:

                  y = validated_df['target']

                  # 假设最后一列是目标变量
                  X = validated_df.iloc[:, :-1]
                  y = validated_df.iloc[:, -1]

              self.feature_names = X.columns.tolist()

              # 数据统计
              logger.info(f"Data loaded: X={{X.shape}}, y={{y.shape}}")
              logger.info(f"Class distribution: {{pd.Series(y).value_counts().to_dict()}}")

              return X, y

              logger.error(f"Data loading failed: {{e}}")

              ('encoder', self._get_encoder(categorical_features))
          ])

          # 组合预处理器
          preprocessor = ColumnTransformer(
              transformers=[
                  ('num', numeric_transformer, numeric_features),
                  ('cat', categorical_transformer, categorical_features)
              ],
              remainder='passthrough'
          )

          self.preprocessor = preprocessor
          return preprocessor

      def _get_imputer(self, feature_type: str):
          """获取插补器"""
          if self.config.imputation_strategy == 'knn':
              return KNNImputer(n_neighbors=5)
          elif feature_type == 'numeric':
              strategy = self.config.imputation_strategy
              return SimpleImputer(strategy=strategy)
          else:
              return SimpleImputer(strategy='most_frequent')

      def _get_scaler(self):
          """获取缩放器"""
          scalers = {{
              'standard': StandardScaler(),
              'robust': RobustScaler(),
              'minmax': MinMaxScaler()
          }}
          return scalers.get(self.config.scaling_method, RobustScaler())

          self.study = None

      def optimize(self, X: pd.DataFrame, y: pd.Series, 
                   preprocessor: ColumnTransformer, n_trials: int = 100):
          """
          使用Optuna进行超参数优化
          
          Args:
              X: 训练特征
              y: 训练标签  
              preprocessor: 预处理器
              n_trials: 优化试验次数
          """
          import optuna
          from optuna.samplers import TPESampler
          from optuna.pruners import MedianPruner

          logger.info("Starting hyperparameter optimization")

          def objective(trial):
              # 根据模型类型定义搜索空间
              params = self._get_param_space(trial)

              # 创建模型
              model = self._create_model_with_params(params)

              # 创建pipeline
              pipeline = Pipeline([
                  ('preprocessor', preprocessor),
                  ('classifier', model)
              ])

              # 交叉验证
              cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
              scores = cross_val_score(
                  pipeline, X, y, cv=cv, scoring='f1_weighted', n_jobs=-1
              )

              return scores.mean()

          # 创建study
          self.study = optuna.create_study(
              direction='maximize',
              sampler=TPESampler(seed=42),
              pruner=MedianPruner(n_startup_trials=10)
          )

          # 优化
          self.study.optimize(objective, n_trials=n_trials)

          self.best_params = self.study.best_params

          logger.info(f"Best parameters: {{self.best_params}}")
          logger.info(f"Best score: {{self.study.best_value:.4f}}")

          }}

                  self.config.model_params.update(best_params)
                  results['best_params'] = best_params

              # Step 5: 模型训练
              logger.info("Step 5: Model training")
              trained_model = self.trainer.train(X_train, y_train, preprocessor)

              logger.info("Step 6: Model evaluation")

              model_uri = self._save_and_register_model(trained_model)

      async def recommend_models(self, features: dict, requirements: dict) -> dict:
          """推荐最适合的模型"""

          # 准备特征统计
          feature_stats = self.calculate_feature_statistics(features)

          # 使用Few-shot增强的Prompt
          enhanced_prompt = self.prompts['few_shot'] + self.prompts['model_recommendation']

          prompt = enhanced_prompt.format(
              feature_analysis=json.dumps(features, indent=2),
              n_samples=feature_stats['n_samples'],
              n_features=feature_stats['n_features'],
              target_type=requirements.get('target_type', 'binary'),
              feature_statistics=json.dumps(feature_stats, indent=2),
              business_goal=requirements.get('goal', '异常检测'),
              performance_requirements=requirements.get('performance', '高准确率'),
              deployment_env=requirements.get('deployment', 'cloud')
          )

          response = await self.llm.complete(prompt)
          return self.parse_json(response)

      async def generate_pipeline_code(self, model_rec: dict, features: dict) -> str:
          """生成完整的Pipeline代码"""

          # 根据推荐模型生成特定的导入和创建代码
          model_name = model_rec['model']
          model_imports, model_creation = self.get_model_specific_code(model_name)

          prompt = self.prompts['code_generation'].format(
              recommended_model=model_name,
              feature_info=json.dumps(features, indent=2),
              data_path='data/security_logs.csv',
              task_type='classification',
              timestamp=pd.Timestamp.now(),
              model_imports=model_imports,
              model_creation_code=model_creation,
              hyperparameter_tuning_code=self.get_tuning_code(model_name),
              enable_tuning='True'
          )

          response = await self.llm.complete(prompt)
          return self.extract_code(response)

      async def generate_optimization_code(self, model_name: str, features: dict) -> str:
          """生成超参数优化代码"""

          prompt = self.prompts['hyperparameter'].format(
              model_type=model_name,
              data_size=f"{features.get('n_samples', 10000)} samples",
              compute_resources="8 CPUs, 32GB RAM",
              time_budget="1 hour"
          )

          response = await self.llm.complete(prompt)
          result = self.parse_json(response)
          return result.get('optimization_code', '')
