  import asyncio
  import json
  from typing import Dict, List, Any
  import pandas as pd
  import numpy as np

  class UnifiedFeatureExtractor:
      """
      统一的LLM特征提取器，支持ML/DL/RL全流程
      """

      def __init__(self, llm_client, config=None):
          self.llm = llm_client
          self.config = config or self.default_config()

          # 加载所有Prompt模板
          self.prompts = {
              'ml_statistical': ML_STATISTICAL_FEATURES_PROMPT,
              'ml_engineered': ML_ENGINEERED_FEATURES_PROMPT,
              'dl_sequence': DL_SEQUENCE_REPRESENTATION_PROMPT,
              'dl_image': DL_IMAGE_REPRESENTATION_PROMPT,
              'dl_graph': DL_GRAPH_REPRESENTATION_PROMPT,
              'rl_mdp': RL_MDP_DEFINITION_PROMPT,
              'rl_policy': RL_POLICY_LEARNING_PROMPT,
              'rl_safe': RL_SAFE_LEARNING_PROMPT,
              'validation': FEATURE_VALIDATION_PROMPT
          }

          self.cache = {}
          self.validation_results = {}

      def default_config(self):
          return {
              'ml': {
                  'feature_types': ['basic', 'engineered'],
                  'scaling': 'robust',
                  'encoding': 'target'
              },
              'dl': {
                  'max_seq_length': 512,
                  'vocab_size': 10000,
                  'embedding_dim': 128,
                  'image_size': 224
              },
              'rl': {
                  'state_dim': 128,
                  'action_dim': 25,
                  'discount_factor': 0.99,
                  'safety_threshold': 0.95
              }
          }

      async def extract_all_features(self, 
                                     behavior_logs: str,
                                     target_models: List[str] = ['ml', 'dl', 'rl'],
                                     validate: bool = True) -> Dict[str, Any]:
          """
          完整的特征提取流程
          
          Args:
              behavior_logs: 原始行为日志
              target_models: 目标模型类型列表
              validate: 是否验证特征质量
              
          Returns:
              包含所有特征的字典
          """

          results = {
              'metadata': {
                  'timestamp': pd.Timestamp.now().isoformat(),
                  'log_size': len(behavior_logs),
                  'target_models': target_models
              },
              'features': {}
          }

          # 并行提取不同类型的特征
          tasks = []

          if 'ml' in target_models:
              tasks.append(self.extract_ml_features(behavior_logs))
          if 'dl' in target_models:
              tasks.append(self.extract_dl_features(behavior_logs))
          if 'rl' in target_models:
              tasks.append(self.extract_rl_features(behavior_logs))

          # 等待所有任务完成
          extracted_features = await asyncio.gather(*tasks)

          # 组织结果
          for i, model_type in enumerate(target_models):
              results['features'][model_type] = extracted_features[i]

          # 验证特征质量
          if validate:
              validation_tasks = []
              for model_type, features in results['features'].items():
                  validation_tasks.append(
                      self.validate_features(features, model_type)
                  )

              validation_results = await asyncio.gather(*validation_tasks)
              results['validation'] = dict(zip(target_models, validation_results))

          # 生成特征报告
          results['report'] = self.generate_feature_report(results)

          return results

      async def extract_ml_features(self, logs: str) -> Dict:
          """提取ML特征"""

          # 阶段1: 基础统计特征
          basic_prompt = self.prompts['ml_statistical'].format(
              log_data=logs
          )
          basic_features = await self.llm_complete(basic_prompt)

          # 阶段2: 工程特征
          # 首先获取基线数据
          baseline = await self.get_user_baseline(logs)
          peer_stats = await self.get_peer_group_stats(logs)

          engineered_prompt = self.prompts['ml_engineered'].format(
              basic_features=basic_features,
              user_baseline=baseline,
              peer_group_stats=peer_stats
          )
          engineered_features = await self.llm_complete(engineered_prompt)

          # 组合特征
          ml_features = {
              'basic': self.parse_json(basic_features),
              'engineered': self.parse_csv(engineered_features),
              'feature_matrix': self.create_feature_matrix(
                  basic_features, engineered_features
              )
          }

          return ml_features

      async def extract_dl_features(self, logs: str) -> Dict:
          """提取DL特征"""

          dl_features = {}

          # 序列表示
          sequence_prompt = self.prompts['dl_sequence'].format(
              behavior_logs=logs,
              max_len=self.config['dl']['max_seq_length'],
              vocab_size=self.config['dl']['vocab_size'],
              embed_dim=self.config['dl']['embedding_dim']
          )
          dl_features['sequences'] = await self.llm_complete(sequence_prompt)

          # 图像表示（可选）
          if self.config.get('dl', {}).get('use_cnn', False):
              image_prompt = self.prompts['dl_image'].format(
                  behavior_sequence=logs,
                  time_window='24h',
                  resolution=self.config['dl']['image_size']
              )
              dl_features['images'] = await self.llm_complete(image_prompt)

          # 图表示（可选）
          if self.config.get('dl', {}).get('use_gnn', False):
              graph_prompt = self.prompts['dl_graph'].format(
                  behavior_data=logs
              )
              dl_features['graphs'] = await self.llm_complete(graph_prompt)

          return dl_features

      async def extract_rl_features(self, logs: str) -> Dict:
          """提取RL特征"""

          # 获取安全约束
          constraints = await self.get_security_constraints()

          # MDP定义
          mdp_prompt = self.prompts['rl_mdp'].format(
              behavior_sequence=logs,
              security_constraints=constraints
          )
          mdp_features = await self.llm_complete(mdp_prompt)

          # 策略学习特征
          policy_prompt = self.prompts['rl_policy'].format(
              trajectory_data=logs,
              expert_demonstrations=await self.get_expert_demos()
          )
          policy_features = await self.llm_complete(policy_prompt)

          # 安全RL特征（可选）
          safe_features = None
          if self.config.get('rl', {}).get('use_safety', True):
              safe_prompt = self.prompts['rl_safe'].format(
                  behavior_data=logs,
                  safety_specifications=constraints
              )
              safe_features = await self.llm_complete(safe_prompt)

          return {
              'mdp': self.parse_json(mdp_features),
              'policy': self.parse_json(policy_features),
              'safety': self.parse_json(safe_features) if safe_features else None
          }

      async def validate_features(self, features: Dict, feature_type: str) -> Dict:
          """验证特征质量"""

          validation_prompt = self.prompts['validation'].format(
              extracted_features=json.dumps(features, indent=2),
              feature_type=feature_type.upper()
          )

          validation_result = await self.llm_complete(validation_prompt)
          return self.parse_json(validation_result)

      async def llm_complete(self, prompt: str) -> str:
          """调用LLM完成prompt"""

          # 检查缓存
          prompt_hash = hash(prompt[:100])  # 使用前100字符作为缓存键
          if prompt_hash in self.cache:
              return self.cache[prompt_hash]

          # 调用LLM
          response = await self.llm.complete(prompt)

          # 缓存结果
          self.cache[prompt_hash] = response

          return response

      def parse_json(self, response: str) -> Dict:
          """解析JSON响应"""
          try:
              # 清理响应（去除markdown标记等）
              cleaned = response.strip()
              if cleaned.startswith('```json'):
                  cleaned = cleaned[7:]
              if cleaned.endswith('```'):
                  cleaned = cleaned[:-3]

              return json.loads(cleaned)
          except json.JSONDecodeError as e:
              print(f"JSON解析错误: {e}")
              return {}

      def parse_csv(self, response: str) -> pd.DataFrame:
          """解析CSV响应"""
          try:
              # 清理响应
              cleaned = response.strip()
              if cleaned.startswith('```csv'):
                  cleaned = cleaned[6:]
              if cleaned.endswith('```'):
                  cleaned = cleaned[:-3]

              # 转换为DataFrame
              from io import StringIO
              return pd.read_csv(StringIO(cleaned))
          except Exception as e:
              print(f"CSV解析错误: {e}")
              return pd.DataFrame()

      def create_feature_matrix(self, basic: str, engineered: str) -> np.ndarray:
          """创建特征矩阵"""
          # 解析特征
          basic_dict = self.parse_json(basic)
          engineered_df = self.parse_csv(engineered)

          # 展平基础特征
          basic_values = []
          for category in basic_dict.values():
              if isinstance(category, dict):
                  for value in category.values():
                      if isinstance(value, (int, float)):
                          basic_values.append(value)

          # 合并工程特征
          if not engineered_df.empty:
              engineered_values = engineered_df['feature_value'].values
              all_features = np.concatenate([basic_values, engineered_values])
          else:
              all_features = np.array(basic_values)

          return all_features.reshape(1, -1)  # 返回行向量

      def generate_feature_report(self, results: Dict) -> str:
          """生成特征提取报告"""

          report = []
          report.append("=" * 50)
          report.append("特征提取报告")
          report.append("=" * 50)

          # 元数据
          report.append(f"\n时间: {results['metadata']['timestamp']}")
          report.append(f"日志大小: {results['metadata']['log_size']} 字符")
          report.append(f"目标模型: {', '.join(results['metadata']['target_models'])}")

          # 特征统计
          report.append("\n特征统计:")
          for model_type, features in results['features'].items():
              report.append(f"\n{model_type.upper()}特征:")
              if isinstance(features, dict):
                  for key, value in features.items():
                      if value is not None:
                          if isinstance(value, dict):
                              report.append(f"  - {key}: {len(value)} 个特征类别")
                          elif isinstance(value, pd.DataFrame):
                              report.append(f"  - {key}: {value.shape} 维度")
                          elif isinstance(value, np.ndarray):
                              report.append(f"  - {key}: {value.shape} 形状")

          # 验证结果
          if 'validation' in results:
              report.append("\n验证结果:")
              for model_type, validation in results['validation'].items():
                  if validation and isinstance(validation, dict):
                      report.append(f"\n{model_type.upper()}:")
                      if 'recommendations' in validation:
                          report.append("  推荐:")
                          for rec_type, recs in validation['recommendations'].items():
                              report.append(f"    - {rec_type}: {len(recs)} 条建议")

          return "\n".join(report)
